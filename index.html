Google Sheets + Drive + Automated Gmail Script

Paste this into Extensions → Apps Script inside your 500-vehicle Google Sheet and set a daily 8:00 AM trigger. Zero server or API fees. 

function checkFleetComplianceAndSendEmails() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Fleet500");
  const data = sheet.getDataRange().getValues();
  const today = new Date();
  const adminEmail = "fleet.compliance@company.in";
  let blockedList = [];
  let expiringList = [];

  for (let i = 1; i < data.length; i++) {
    const regNo = data[i][0];
    const pucExpiry = new Date(data[i][4]);
    const insExpiry = new Date(data[i][5]);
    const minExpiry = new Date(Math.min(pucExpiry, insExpiry));
    const daysLeft = Math.ceil((minExpiry - today) / (1000 * 60 * 60 * 24));

    if (daysLeft < 0) {
      sheet.getRange(i + 1, 8).setValue("BLOCKED");
      blockedList.push(regNo + " (Expired " + Math.abs(daysLeft) + "d ago)");
    } else if (daysLeft <= 30) {
      sheet.getRange(i + 1, 8).setValue("EXPIRING_SOON");
      expiringList.push(regNo + " (" + daysLeft + " days left)");
    } else {
      sheet.getRange(i + 1, 8).setValue("CLEARED");
    }
  }

  if (blockedList.length > 0 || expiringList.length > 0) {
    const body = "FleetGate Daily Compliance Summary:\n\n" +
      "RED GATE-BLOCKED VEHICLES (" + blockedList.length + "):\n" + blockedList.join("\n") +
      "\n\nAMBER EXPIRING SOON (" + expiringList.length + "):\n" + expiringList.join("\n") +
      "\n\nWhatsApp Escalation Contact: +91 7760914701";
    MailApp.sendEmail(adminEmail, "FleetGate Alert: " + blockedList.length + " Vehicles Blocked at Gate", body);
  }
}
